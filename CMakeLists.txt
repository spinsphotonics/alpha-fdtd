cmake_minimum_required(VERSION 3.24)
# A CC or CXX naming a compiler that is not installed (some Python builds export CC=clang to pip) falls back to the default.
foreach(v CC CXX)
  if(DEFINED ENV{${v}} AND NOT "$ENV{${v}}" MATCHES " ")
    find_program(_afdtd_${v} "$ENV{${v}}")
    if(NOT _afdtd_${v})
      message(WARNING "${v}=$ENV{${v}} is not installed; using the default compiler")
      unset(ENV{${v}})
    endif()
  endif()
endforeach()
project(alpha-fdtd VERSION 0.2.0 LANGUAGES C CXX CUDA)

# Architectures to build, from 80 (A100), 90a (H100, H200), 100a (B200), 103a (B300).  Each needs its own SASS: the "a" targets have no
# PTX fallback, so a GPU without an object is refused at plan time instead of running a JIT-compiled, unverified kernel.
set(AFDTD_ARCHS "80;90a;100a;103a" CACHE STRING "GPU architectures to compile the kernel objects for")
option(AFDTD_BUILD_TESTS "Build the tests in tests/" OFF)
option(AFDTD_BUILD_TOOLS "Build tools/bench (developer benchmark)" OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_CUDA_VISIBILITY_PRESET hidden)
find_package(CUDAToolkit REQUIRED)
find_package(Threads REQUIRED)

# ---- kernel objects: one compilation of src/object.cu per (kernel text, architecture)
set(_objs "")
set(_inc "")
foreach(arch IN LISTS AFDTD_ARCHS)
  if(arch STREQUAL "80")
    set(kernels zs7_cpasync_l2h0 zs7_cpasync_l2h3)
    set(cc 80)
  elseif(arch MATCHES "^(90a|100a|103a)$")
    set(kernels zs7_tma)
    string(REGEX REPLACE "a$" "" cc "${arch}")
  else()
    message(FATAL_ERROR "AFDTD_ARCHS: unsupported architecture '${arch}' (80, 90a, 100a, 103a)")
  endif()
  foreach(k IN LISTS kernels)
    foreach(prec bf16 f16)                       # each text in BF16 and in FP16 (-DAFDTD_FP16: the same text with __half2)
      set(t afdtd_obj_${k}_sm${arch}_${prec})
      add_library(${t} OBJECT src/object.cu)
      set_target_properties(${t} PROPERTIES CUDA_ARCHITECTURES OFF)
      target_include_directories(${t} PRIVATE src)
      target_compile_definitions(${t} PRIVATE KERN="kernels/${k}.cuh" AFDTD_NS=${t}_ns AFDTD_SYM=${t} AFDTD_NAME="${k}"
                                 AFDTD_ARCH="sm_${arch}" AFDTD_CC=${cc})
      if(prec STREQUAL "f16")
        target_compile_definitions(${t} PRIVATE AFDTD_FP16=1)
      endif()
      # 177, 550: variables the generated kernel texts declare and do not use
      target_compile_options(${t} PRIVATE -gencode=arch=compute_${arch},code=sm_${arch} -O3 -diag-suppress=177,550)
      list(APPEND _objs $<TARGET_OBJECTS:${t}>)
      string(APPEND _inc "AFDTD_OBJECT(${t})\n")
    endforeach()
  endforeach()
endforeach()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/gen/objects.inc "${_inc}")

# kernel-set id for wisdom: a hash of every source that decides what the kernels compute and how fast
set(_kset "")
foreach(f src/object.cu src/internal.hpp src/kernels/zs7_tma.cuh src/kernels/zs7_cpasync_l2h0.cuh src/kernels/zs7_cpasync_l2h3.cuh)
  file(SHA1 ${CMAKE_CURRENT_SOURCE_DIR}/${f} h)
  string(APPEND _kset ${h})
endforeach()
string(SHA1 _kset "${_kset}")
string(SUBSTRING ${_kset} 0 12 AFDTD_KSET)

# ---- the FP32 executor, injectors and the DFT of recordings (src/f32.cu, src/inject.cu, src/dft.cu): plain CUDA, SASS per family and PTX
set(_plain_archs "")
foreach(arch IN LISTS AFDTD_ARCHS)
  string(REGEX REPLACE "a$" "" a "${arch}")
  list(APPEND _plain_archs ${a}-real)
endforeach()
add_library(afdtd_dft OBJECT src/dft.cu src/inject.cu src/f32.cu)
set_target_properties(afdtd_dft PROPERTIES CUDA_ARCHITECTURES "${_plain_archs};80-virtual")
target_include_directories(afdtd_dft PRIVATE src include)
target_compile_options(afdtd_dft PRIVATE -O3)

set(_host src/api.cpp src/plan.cpp src/boxio.cpp src/record.cpp)
add_library(alphafdtd SHARED ${_host} ${_objs} $<TARGET_OBJECTS:afdtd_dft>)
target_include_directories(alphafdtd PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>
                           PRIVATE src ${CMAKE_CURRENT_BINARY_DIR}/gen)
target_compile_definitions(alphafdtd PRIVATE AFDTD_KSET="${AFDTD_KSET}")
target_link_libraries(alphafdtd PRIVATE CUDA::cudart_static Threads::Threads ${CMAKE_DL_LIBS})
target_link_options(alphafdtd PRIVATE -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map -static-libstdc++ -static-libgcc)
set_target_properties(alphafdtd PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)

if(AFDTD_BUILD_TOOLS)
  add_executable(afdtd_bench tools/bench/bench.c)
  target_link_libraries(afdtd_bench PRIVATE alphafdtd)
endif()

# ---- tests: tests/CMakeLists.txt registers its tests with afdtd_add_test(name COMMAND ... LABELS ...)
function(afdtd_add_test name)
  cmake_parse_arguments(A "" "" "COMMAND;LABELS" ${ARGN})
  add_test(NAME ${name} COMMAND ${A_COMMAND})
  set_tests_properties(${name} PROPERTIES LABELS "${A_LABELS}" SKIP_RETURN_CODE 77)
endfunction()
if(AFDTD_BUILD_TESTS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/CMakeLists.txt)
  enable_testing()
  add_subdirectory(tests)
endif()

install(FILES include/alpha_fdtd.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT alpha-fdtd NAMESPACE alpha-fdtd:: FILE alpha-fdtd-config.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/alpha-fdtd)
