# Tests, registered with afdtd_add_test (labels: cpu = no GPU needed; gpu = one supported GPU; gpu2 = two).
# Each test prints one JSON line; exit 77 = skipped (too few GPUs).  Run: ctest --test-dir build -L cpu (or -L gpu ...).
set(_strict -Wall -Wextra -pedantic -Werror)

# ---- CPU
add_executable(afdtd_api_v1 unit/api_v1.c)
set_target_properties(afdtd_api_v1 PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(afdtd_api_v1 PRIVATE ${PROJECT_SOURCE_DIR}/include)
target_compile_options(afdtd_api_v1 PRIVATE ${_strict})
add_executable(afdtd_api_v1_cxx unit/api_v1_cxx.cpp)
target_include_directories(afdtd_api_v1_cxx PRIVATE ${PROJECT_SOURCE_DIR}/include unit)
target_compile_options(afdtd_api_v1_cxx PRIVATE ${_strict})
afdtd_add_test(api_v1 COMMAND afdtd_api_v1 ${CMAKE_CURRENT_SOURCE_DIR}/data/api_v1.txt LABELS cpu)
afdtd_add_test(api_v1_cxx COMMAND afdtd_api_v1_cxx ${CMAKE_CURRENT_SOURCE_DIR}/data/api_v1.txt LABELS cpu)

# ---- GPU: exact against tests/oracle/reference.cu
add_executable(afdtd_exact gpu/exact.cu)
set(_oracle_archs "")                       # the oracle needs no "a" features: plain SASS per family, and PTX
foreach(arch IN LISTS AFDTD_ARCHS)
  string(REGEX REPLACE "a$" "" a "${arch}")
  list(APPEND _oracle_archs ${a}-real)
endforeach()
set_target_properties(afdtd_exact PROPERTIES CUDA_ARCHITECTURES "${_oracle_archs};80-virtual")
target_include_directories(afdtd_exact PRIVATE oracle)
target_compile_options(afdtd_exact PRIVATE -O3)
target_link_libraries(afdtd_exact PRIVATE alphafdtd)

function(exact name labels)
  afdtd_add_test(exact_${name} COMMAND afdtd_exact ${ARGN} LABELS ${labels})
endfunction()
